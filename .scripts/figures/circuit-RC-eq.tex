\documentclass[border=8pt]{standalone}
\usepackage{fontspec}
\setmainfont{Cambria}
\usepackage{unicode-math}
\setmathfont{Cambria Math}
\usepackage[straightvoltages]{circuitikz}

\ctikzset{bipoles/thickness=1.35}   % 元件线宽，与正文图线统一

\begin{document}
\begin{circuitikz}[scale=1.2, transform shape]
  \draw (0,0)
    to[sinusoidal voltage source, l=$u_r(t)$] (0,3)
    to[R, l=$R$, i=$i$]                       (3.2,3)
    to[C, l=$C$, v=$u_c$, voltage=straight]   (3.2,0)
    -- (0,0);
\end{circuitikz}
\end{document}
